-- update.qql — UPDATE statements

-- Update payload by ID
UPDATE docs SET PAYLOAD = {status: 'reviewed'} WHERE id = 42;

-- Update payload by filter
UPDATE docs SET PAYLOAD = {status: 'reviewed'} WHERE status = 'pending';

-- Update payload with multiple fields
UPDATE docs SET PAYLOAD = {status: 'reviewed', reviewer: 'admin'} WHERE id = 42;

-- Update vector by ID
UPDATE docs SET VECTOR = [0.1, 0.2, 0.3] WHERE id = 42;

-- Update specific named vector
UPDATE docs SET VECTOR dense = [0.1, 0.2, 0.3] WHERE id = 42;

-- Update vector with sparse vector value
UPDATE docs SET VECTOR sparse = {indices: [0, 3], values: [0.5, 0.8]} WHERE id = 42;

-- Update with string UUID
UPDATE docs SET PAYLOAD = {status: 'updated'} WHERE id = 'a1b2c3d4';
