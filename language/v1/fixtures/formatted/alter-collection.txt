-- alter-collection.qql — ALTER COLLECTION statements

-- Alter vector config
ALTER COLLECTION docs WITH VECTOR (on_disk = true);

-- Alter a named dense vector diff
ALTER COLLECTION docs WITH VECTOR dense (HNSW (m = 32), QUANTIZATION (type = 'scalar', always_ram = true), VECTOR (memory = 'cold'));

-- Disable per-vector quantization
ALTER COLLECTION docs WITH VECTOR colbert (QUANTIZATION (disabled = true));

-- Alter a named sparse vector diff
ALTER COLLECTION docs WITH SPARSE bm25 (SPARSE (modifier = 'idf', full_scan_threshold = 5000, datatype = 'float16'));

-- Alter HNSW
ALTER COLLECTION docs WITH HNSW (m = 32);

-- Alter with OPTIMIZERS
ALTER COLLECTION docs WITH OPTIMIZERS (max_segment_size = 500000);

-- Alter with PARAMS
ALTER COLLECTION docs WITH PARAMS (replication_factor = 3);

-- Alter with QUANTIZATION
ALTER COLLECTION docs WITH QUANTIZATION (type = 'scalar');

-- Disable quantization
ALTER COLLECTION docs WITH QUANTIZATION (disabled = true);
