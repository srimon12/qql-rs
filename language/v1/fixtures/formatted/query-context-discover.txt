-- query-context-discover.qql — Context and Discover queries

-- Context query
QUERY CONTEXT (POSITIVE POINT 1 NEGATIVE POINT 2) FROM docs LIMIT 10;

-- Context with multiple pairs
QUERY CONTEXT (POSITIVE POINT 1 NEGATIVE POINT 2, POSITIVE POINT 3 NEGATIVE POINT 4)
FROM docs
LIMIT 10;

-- Context with query inputs
QUERY CONTEXT (POSITIVE 'good' NEGATIVE 'bad') FROM docs LIMIT 10;

-- Discover query
QUERY DISCOVER TARGET POINT 1 CONTEXT (POSITIVE POINT 2 NEGATIVE POINT 3)
FROM docs
LIMIT 10;

-- Discover with multiple context pairs
QUERY DISCOVER TARGET 'search' CONTEXT (POSITIVE POINT 1 NEGATIVE POINT 2, POSITIVE POINT 3 NEGATIVE POINT 4)
FROM docs
LIMIT 10;
